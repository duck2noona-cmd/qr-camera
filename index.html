// 공유하기 / 저장 호환 ZIP 다운로드 함수 async function downloadAllAsZip() { if (capturedPhotos.length === 0) return; statusText.innerText = "압축 파일 생성 중..."; zipBtn.disabled = true; const zip = new JSZip(); capturedPhotos.forEach((photo) => { const base64Data = photo.dataUrl.split(',')[1]; zip.file(photo.fileName, base64Data, { base64: true }); }); try { const content = await zip.generateAsync({ type: "blob" }); const fileName = `QR_Photos_${new Date().getTime()}.zip`; const file = new File([content], fileName, { type: "application/zip" }); // 브라우저가 직접 공유(Web Share API)를 지원하는 경우 (갤럭시/아이폰 최신 브라우저) if (navigator.canShare && navigator.canShare({ files: [file] })) { await navigator.share({ files: [file], title: 'QR 사진 모음', text: 'QR 인식 사진 ZIP 파일입니다.' }); statusText.innerText = "공유가 완료되었습니다!"; } else { // 미지원 시 기존 파일 저장 방식 작동 saveAs(content, fileName); statusText.innerText = "ZIP 다운로드가 시작되었습니다!"; } } catch (err) { // 사용자가 공유 창을 취소했을 때 등의 예외 처리 if (err.name !== 'AbortError') { statusText.innerText = "저장/공유 중 오류가 발생했습니다."; console.error(err); } else { statusText.innerText = "공유가 취소되었습니다."; } } finally { zipBtn.disabled = false; } }
